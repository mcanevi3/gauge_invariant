\section{The System Model}
A linear time-invariant discrete-time system governed by the state-space equations:
\begin{align}
    x_{k} &= A x_{k-1} + B_u u_{k-1} + w_{k-1} \\
    y_k &= C x_k + v_k
\end{align}
where:
\begin{itemize}
    \item $x_k \in \mathbb{R}^n$ is the state vector.
    \item $u_k \in \mathbb{R}^m$ is the control input vector.
    \item $y_k \in \mathbb{R}^p$ is the measured output.
    \item $w_k \in \mathbb{R}^n$ is the process noise with covariance $Q = E[w_k w_k^T]$.
    \item $v_k \in \mathbb{R}^p$ is the measurement noise with covariance matrix $R = E[v_k v_k^T]$, independent of $w_k$.
\end{itemize}

\section{The Kalman Filter}
The filter is defined as:
\begin{align}
    \hat{x}_{k|k-1} &= A \hat{x}_{k-1|k-1} + B_u u_{k-1} 
\end{align}
The error dynamics associated is obtained as follows,
\begin{align}
    e_{k|k-1} &= x_{k}-\hat{x}_{k|k-1}\\
    &=A x_{k-1} + B_u u_{k-1} + w_{k-1}-A \hat{x}_{k-1|k-1} - B_u u_{k-1} \\
    e_{k|k-1} &=A e_{k-1|k-1}+w_{k-1}
\end{align}
The covariance matrix $P_{k|k-1}$ is given by $E[e_{k|k-1}e_{k|k-1}^T]$ as follows,
\begin{align}
    P_{k|k-1} &=E[e_{k|k-1}e_{k|k-1}^T]\\
    &=E[(A e_{k-1|k-1}+w_{k-1})(A e_{k-1|k-1}+w_{k-1})^T]\\
    &=E[A e_{k-1|k-1} e_{k-1|k-1}^T A^T] + E[A e_{k-1|k-1} w_{k-1}^T] \\
    &+ E[w_{k-1} e_{k-1|k-1}^T] + E[w_{k-1} w_{k-1}^T]\\
    &=A E[e_{k-1|k-1} e_{k-1|k-1}^T] A^T + E[w_{k-1} w_{k-1}^T]\\
    P_{k|k-1} &=A P_{k-1|k-1} A^T + Q
\end{align}
The state estimation is corrected in a Luenberger fashion as follows,
\begin{align}
    \hat{x}_{k|k} &= \hat{x}_{k|k-1} + K_k \left( y_k - C \hat{x}_{k|k-1} \right)
\end{align}
where the error is obtained as,
\begin{align}
    e_{k|k} &= x_k - \hat{x}_{k|k}\\
    &=x_k - \hat{x}_{k|k-1} - K_k \left( y_k - C \hat{x}_{k|k-1} \right)\\
    &=x_k - \hat{x}_{k|k-1} - K_k \left(  C x_k + v_k - C \hat{x}_{k|k-1} \right)\\
    &=e_{k|k-1} - K_k C e_{k|k-1}-K_k v_k\\
    e_{k|k} &= (I-K_k C) e_{k|k-1} - K_k v_k
\end{align}
and its covariance is calculated as,
\begin{align}
    P_{k|k} &= E[e_{k|k} e_{k|k}^T]\\
    &=E[((I-K_k C) e_{k|k-1} - K_k v_k )(e_{k|k-1}(I-K_k C)^T  - v_k^T K_k^T)]\\
    &=E[(I-K_k C) e_{k|k-1}e_{k|k-1}(I-K_k C)^T]-E[(I-K_k C) e_{k|k-1}v_k^T K_k^T]\\
    &-E[K_k v_k e_{k|k-1}^T(I-K_k C)^T]+E[K_k v_k v_k^T K_k^T ]\\
    &=(I-K_k C) E[e_{k|k-1}e_{k|k-1}](I-K_k C)^T-(I-K_k C)E[e_{k|k-1}v_k^T]K_k^T\\
    &-K_k E[v_k e_{k|k-1}^T](I-K_k C)^T+K_kE[v_k v_k^T] K_k^T\\
    &=(I-K_k C) E[e_{k|k-1}e_{k|k-1}](I-K_k C)^T+K_kE[v_k v_k^T] K_k^T\\
    P_{k|k} &=(I-K_k C) P_{k|k-1} (I-K_k C)^T+K_k R K_k^T
\end{align}
The covariance equation is expanded as follows,
\begin{align}
    P_{k|k} &=(I-K_k C) P_{k|k-1} (I-K_k C)^T+K_k R K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-P_{k|k-1} C^T K_k^T-K_k C P_{k|k-1} C^T K_k^T+K_k R K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-P_{k|k-1} C^T K_k^T+K_k (C P_{k|k-1} C^T+R) K_k^T
\end{align}
This equation is used to choose the Kalman gain $K_k$ with the following problem,
\begin{equation}
    \min_{K_k} \text{Tr}(P_{k|k})
\end{equation}
by utilizing 
\begin{equation}
    \frac{\partial \text{Tr}(AB)}{\partial A} = B^T
\end{equation}
and
\begin{equation}
    \frac{\partial \text{Tr}(ACA^T)}{\partial A} = 2AC
\end{equation}
The following is obtained,
\begin{align}
    \frac{\partial \text{Tr}(P_{k|k})}{\partial K_k} &= -2 \left( P_{k|k-1} C^T \right)^T + 2 K_k \left( C P_{k|k-1} C^T + R \right)\\
    &=0
\end{align}
which gives,
\begin{align}
    2 \left( P_{k|k-1} C^T \right) &= 2 K_k \left( C P_{k|k-1} C^T + R \right)\\
    P_{k|k-1} C^T  &= K_k (C P_{k|k-1} C^T + R) \\
    P_{k|k-1} C^T(C P_{k|k-1} C^T + R)^{-1}  &= K_k \\
    K_k&= P_{k|k-1} C^T(C P_{k|k-1} C^T + R)^{-1}\\
    K_k&= P_{k|k-1} C^T S_k^{-1}
\end{align}
where $S_k = C P_{k|k-1}C^T+R$.

The covariance equation 
\begin{align}
    P_{k|k} &=(I-K_k C) P_{k|k-1} (I-K_k C)^T+K_k R K_k^T
\end{align}
using $K_k S_k=P_{k|k-1} C^T$ when examined gives,
\begin{align}
    P_{k|k} &=(I-K_k C) P_{k|k-1} (I-K_k C)^T+K_k R K_k^T\\
    &=(P_{k|k-1} -K_k C P_{k|k-1} )(I-C^T K_k^T)+K_k R K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-P_{k|k-1}C^T K_k^T+K_k C P_{k|k-1}C^T K_k^T+K_k R K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-P_{k|k-1}C^T K_k^T+K_k (C P_{k|k-1}C^T+R) K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-(K_k S_k) K_k^T+K_k S_k K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-K_k S_k K_k^T+K_k S_k K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}\\
    P_{k|k} &=(I-K_k C) P_{k|k-1}
\end{align}

\section{Summary}
With initial choices of $\hat{x}_0$, $P_0$, the Kalman filter is summarized as follows:
\begin{align}
    \hat{x}_{k|k-1} &= A \hat{x}_{k-1|k-1} + B_u u_{k-1}
\end{align}
the filter is:
\begin{align}
    S_k &= C P_{k|k-1} C^T + R\\
    K_k &= P_{k|k-1} C^T S_k^{-1}
\end{align}
then,
\begin{align}
    \hat{x}_{k|k} &= \hat{x}_{k|k-1} + K_k \left( y_k - C \hat{x}_{k|k-1} \right)\\
    P_{k|k} &= (I-K_k C) P_{k|k-1}
\end{align}


\section{Two-Stage Estimation Approach}
In the presence of an unknown disturbance $w_k$, the discrete-time measurement equation is given by:
\begin{equation}
    y_k = C x_k + D_w w_k + v_k
\end{equation}
To decouple the state $x_k$ from the disturbance $w_k$ during the initial estimation phase, a projection matrix $\Pi$ is defined such that it nullifies the measurement matrix $C$. Assuming $C$ has full column rank, $\Pi$ is computed via the pseudo-inverse as:
\begin{equation}
    \Pi = I - C (C^T C)^{-1} C^T
\end{equation}
which satisfies $\Pi C = 0$. 

\subsection{Stage 1: Disturbance Estimation}
Multiplying the measurement equation by the projection matrix yields the projected measurement $\bar{y}_k$:
\begin{align}
    \bar{y}_k &= \Pi y_k \\
    &= \Pi C x_k + \Pi D_w w_k + \Pi v_k \\
    &= \Pi D_w w_k + \bar{v}_k
\end{align}
where $\bar{v}_k = \Pi v_k$ is the projected measurement noise with covariance $\bar{R} = \Pi R \Pi^T$. This projection completely removes the influence of the state $x_k$. Assuming the disturbance evolves as a random walk model ($w_k = w_{k-1} + w_{w,k-1}$ with process noise covariance $Q_w$), the Kalman filter for the disturbance is formulated as follows.

The prediction step for the disturbance is:
\begin{align}
    \hat{w}_{k|k-1} &= \hat{w}_{k-1|k-1} \\
    P^w_{k|k-1} &= P^w_{k-1|k-1} + Q_w
\end{align}
The update step uses the projected measurement:
\begin{align}
    S^w_k &= (\Pi D_w) P^w_{k|k-1} (\Pi D_w)^T + \bar{R} \\
    K^w_k &= P^w_{k|k-1} (\Pi D_w)^T (S^w_k)^{-1} \\
    \hat{w}_{k|k} &= \hat{w}_{k|k-1} + K^w_k \left( \Pi y_k - \Pi D_w \hat{w}_{k|k-1} \right) \\
    P^w_{k|k} &= (I - K^w_k \Pi D_w) P^w_{k|k-1}
\end{align}

\subsection{Stage 2: State Estimation}
With the updated disturbance estimate $\hat{w}_{k|k}$ acquired, it is treated as a known input for the primary state estimator. The state prediction relies on the prior disturbance estimate $\hat{w}_{k-1|k-1}$:
\begin{align}
    \hat{x}_{k|k-1} &= A_d \hat{x}_{k-1|k-1} + B_{ud} u_{k-1} + B_{wd} \hat{w}_{k-1|k-1} \\
    P^x_{k|k-1} &= A_d P^x_{k-1|k-1} A_d^T + Q_x
\end{align}
For the state update, the effect of the newly estimated disturbance is subtracted from the actual measurement, creating a modified innovation term. The state update is executed as:
\begin{align}
    S^x_k &= C P^x_{k|k-1} C^T + R \\
    K^x_k &= P^x_{k|k-1} C^T (S^x_k)^{-1} \\
    \hat{x}_{k|k} &= \hat{x}_{k|k-1} + K^x_k \left( y_k - C \hat{x}_{k|k-1} - D_w \hat{w}_{k|k} \right) \\
    P^x_{k|k} &= (I - K^x_k C) P^x_{k|k-1}
\end{align}
This sequential architecture ensures that the disturbance is estimated without interference from the system states, providing a clean feedforward term for the state estimator's correction phase.

\section{Conclusion}
The cross-covariance helps the augmented kalman filter to outperform dual PSM method, PSM for disturbance and PSM for state estimation, but if the disturbance has some discontinuities then the peaks are isolated in dual PSM, hence it performs better.

The augmented kalman can be improved in terms of state estimation by sacrificing the disturbance estimation entirely, such that dual PSM like peak reduction occurs in the states. But disturbance is not estimated at all.

It seems like, the Normalized Innovation Squared signal can be utilized to switch between dual PSM and augmented kalman to improve the overall estimation.
