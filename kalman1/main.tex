\section{The System Model}
A linear time-invariant discrete-time system governed by the state-space equations:
\begin{align}
    x_{k} &= A x_{k-1} + B_u u_{k-1} + w_{k-1} \\
    y_k &= C x_k + v_k
\end{align}
where:
\begin{itemize}
    \item $x_k \in \mathbb{R}^n$ is the state vector.
    \item $u_k \in \mathbb{R}^m$ is the control input vector.
    \item $y_k \in \mathbb{R}^p$ is the measured output.
    \item $w_k \in \mathbb{R}^n$ is the process noise with covariance $Q = E[w_k w_k^T]$.
    \item $v_k \in \mathbb{R}^p$ is the measurement noise with covariance matrix $R = E[v_k v_k^T]$, independent of $w_k$.
\end{itemize}

\section{The Kalman Filter}
The filter is defined as:
\begin{align}
    \hat{x}_{k|k-1} &= A \hat{x}_{k-1|k-1} + B_u u_{k-1} 
\end{align}
The error dynamics associated is obtained as follows,
\begin{align}
    e_{k|k-1} &= x_{k}-\hat{x}_{k|k-1}\\
    &=A x_{k-1} + B_u u_{k-1} + w_{k-1}-A \hat{x}_{k-1|k-1} - B_u u_{k-1} \\
    e_{k|k-1} &=A e_{k-1|k-1}+w_{k-1}
\end{align}
The covariance matrix $P_{k|k-1}$ is given by $E[e_{k|k-1}e_{k|k-1}^T]$ as follows,
\begin{align}
    P_{k|k-1} &=E[e_{k|k-1}e_{k|k-1}^T]\\
    &=E[(A e_{k-1|k-1}+w_{k-1})(A e_{k-1|k-1}+w_{k-1})^T]\\
    &=E[A e_{k-1|k-1} e_{k-1|k-1}^T A^T] + E[A e_{k-1|k-1} w_{k-1}^T] \\
    &+ E[w_{k-1} e_{k-1|k-1}^T] + E[w_{k-1} w_{k-1}^T]\\
    &=A E[e_{k-1|k-1} e_{k-1|k-1}^T] A^T + E[w_{k-1} w_{k-1}^T]\\
    P_{k|k-1} &=A P_{k-1|k-1} A^T + Q
\end{align}
The state estimation is corrected in a Luenberger fashion as follows,
\begin{align}
    \hat{x}_{k|k} &= \hat{x}_{k|k-1} + K_k \left( y_k - C \hat{x}_{k|k-1} \right)
\end{align}
where the error is obtained as,
\begin{align}
    e_{k|k} &= x_k - \hat{x}_{k|k}\\
    &=x_k - \hat{x}_{k|k-1} - K_k \left( y_k - C \hat{x}_{k|k-1} \right)\\
    &=x_k - \hat{x}_{k|k-1} - K_k \left(  C x_k + v_k - C \hat{x}_{k|k-1} \right)\\
    &=e_{k|k-1} - K_k C e_{k|k-1}-K_k v_k\\
    e_{k|k} &= (I-K_k C) e_{k|k-1} - K_k v_k
\end{align}
and its covariance is calculated as,
\begin{align}
    P_{k|k} &= E[e_{k|k} e_{k|k}^T]\\
    &=E[((I-K_k C) e_{k|k-1} - K_k v_k )(e_{k|k-1}(I-K_k C)^T  - v_k^T K_k^T)]\\
    &=E[(I-K_k C) e_{k|k-1}e_{k|k-1}(I-K_k C)^T]-E[(I-K_k C) e_{k|k-1}v_k^T K_k^T]\\
    &-E[K_k v_k e_{k|k-1}^T(I-K_k C)^T]+E[K_k v_k v_k^T K_k^T ]\\
    &=(I-K_k C) E[e_{k|k-1}e_{k|k-1}](I-K_k C)^T-(I-K_k C)E[e_{k|k-1}v_k^T]K_k^T\\
    &-K_k E[v_k e_{k|k-1}^T](I-K_k C)^T+K_kE[v_k v_k^T] K_k^T\\
    &=(I-K_k C) E[e_{k|k-1}e_{k|k-1}](I-K_k C)^T+K_kE[v_k v_k^T] K_k^T\\
    P_{k|k} &=(I-K_k C) P_{k|k-1} (I-K_k C)^T+K_k R K_k^T
\end{align}
The covariance equation is expanded as follows,
\begin{align}
    P_{k|k} &=(I-K_k C) P_{k|k-1} (I-K_k C)^T+K_k R K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-P_{k|k-1} C^T K_k^T-K_k C P_{k|k-1} C^T K_k^T+K_k R K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-P_{k|k-1} C^T K_k^T+K_k (C P_{k|k-1} C^T+R) K_k^T
\end{align}
This equation is used to choose the Kalman gain $K_k$ with the following problem,
\begin{equation}
    \min_{K_k} \text{Tr}(P_{k|k})
\end{equation}
by utilizing 
\begin{equation}
    \frac{\partial \text{Tr}(AB)}{\partial A} = B^T
\end{equation}
and
\begin{equation}
    \frac{\partial \text{Tr}(ACA^T)}{\partial A} = 2AC
\end{equation}
The following is obtained,
\begin{align}
    \frac{\partial \text{Tr}(P_{k|k})}{\partial K_k} &= -2 \left( P_{k|k-1} C^T \right)^T + 2 K_k \left( C P_{k|k-1} C^T + R \right)\\
    &=0
\end{align}
which gives,
\begin{align}
    2 \left( P_{k|k-1} C^T \right) &= 2 K_k \left( C P_{k|k-1} C^T + R \right)\\
    P_{k|k-1} C^T  &= K_k (C P_{k|k-1} C^T + R) \\
    P_{k|k-1} C^T(C P_{k|k-1} C^T + R)^{-1}  &= K_k \\
    K_k&= P_{k|k-1} C^T(C P_{k|k-1} C^T + R)^{-1}\\
    K_k&= P_{k|k-1} C^T S_k^{-1}
\end{align}
where $S_k = C P_{k|k-1}C^T+R$.

The covariance equation 
\begin{align}
    P_{k|k} &=(I-K_k C) P_{k|k-1} (I-K_k C)^T+K_k R K_k^T
\end{align}
using $K_k S_k=P_{k|k-1} C^T$ when examined gives,
\begin{align}
    P_{k|k} &=(I-K_k C) P_{k|k-1} (I-K_k C)^T+K_k R K_k^T\\
    &=(P_{k|k-1} -K_k C P_{k|k-1} )(I-C^T K_k^T)+K_k R K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-P_{k|k-1}C^T K_k^T+K_k C P_{k|k-1}C^T K_k^T+K_k R K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-P_{k|k-1}C^T K_k^T+K_k (C P_{k|k-1}C^T+R) K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-(K_k S_k) K_k^T+K_k S_k K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}-K_k S_k K_k^T+K_k S_k K_k^T\\
    &=P_{k|k-1}-K_k C P_{k|k-1}\\
    P_{k|k} &=(I-K_k C) P_{k|k-1}
\end{align}

\section{Summary}
With initial choices of $\hat{x}_0$, $P_0$, the Kalman filter is summarized as follows:
\begin{align}
    \hat{x}_{k|k-1} &= A \hat{x}_{k-1|k-1} + B_u u_{k-1}
\end{align}
the filter is:
\begin{align}
    S_k &= C P_{k|k-1} C^T + R\\
    K_k &= P_{k|k-1} C^T S_k^{-1}
\end{align}
then,
\begin{align}
    \hat{x}_{k|k} &= \hat{x}_{k|k-1} + K_k \left( y_k - C \hat{x}_{k|k-1} \right)\\
    P_{k|k} &= (I-K_k C) P_{k|k-1}
\end{align}
